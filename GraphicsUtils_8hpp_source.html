124[[nodiscard]] constexpr float calculateRenderDistance(int loadDistanceChunks) {
125 // sqrt(3) ≈ 1.732051 - diagonal distance factor for cube
126 // +0.5 accounts for chunk center offset from corner
127 // +0.1 adds small margin for numerical stability
128 const float diagonalDistance = 1.732051F * (static_cast<float>(loadDistanceChunks) + 0.5F);
129 const float margin = 0.1F;
130 return (diagonalDistance + margin) * Chunk::SIZE_FLOAT;
131}
